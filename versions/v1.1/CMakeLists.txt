cmake_minimum_required(VERSION 3.16)
project(binary_passcode_terminal LANGUAGES CXX)
if(ANDROID)
  add_library(binary_passcode_terminal SHARED binary_passcode_terminal.cpp)
  target_link_options(binary_passcode_terminal PRIVATE "-Wl,-z,max-page-size=16384")
else()
  add_executable(binary_passcode_terminal binary_passcode_terminal.cpp)
  if(WIN32)
    enable_language(RC)
    target_sources(binary_passcode_terminal PRIVATE binary_passcode_terminal.rc)
  endif()
endif()
target_compile_features(binary_passcode_terminal PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(binary_passcode_terminal PRIVATE /utf-8 /W4)
  set_property(TARGET binary_passcode_terminal PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(WIN32)
  target_link_libraries(binary_passcode_terminal PRIVATE bcrypt crypt32)
elseif(NOT ANDROID)
  target_link_libraries(binary_passcode_terminal PRIVATE ${CMAKE_DL_LIBS})
endif()
